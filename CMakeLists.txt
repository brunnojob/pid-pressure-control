cmake_minimum_required(VERSION 3.20)
project(pid_pressure_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pressure_control src/pid.cpp src/controller.cpp)
target_include_directories(pressure_control PUBLIC include)
target_compile_options(pressure_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(pressure_demo app/main.cpp)
target_link_libraries(pressure_demo PRIVATE pressure_control)
enable_testing()
add_executable(pid_tests tests/pid_test.cpp)
target_link_libraries(pid_tests PRIVATE pressure_control)
add_test(NAME pid_tests COMMAND pid_tests)
